api_caller ​

国土交通DPFのAPIをコールし、レスポンスの取得を行います。
すべてのサンプルプログラムでAPIを使用するため、一つのソースファイルで汎用化しています。
API_KEY の値は、ご自身が取得したキーが必要です。
不明な方は、APIキーを確認する をご確認下さい。
APIキーを指定しない場合、または存在しないキーを指定した場合は、ステータス401（Unauthorized）が返されます。

js
// 通信に必要なライブラリのインポート
import axios from 'axios';

// エンドポイントとAPIキーを定義しておく
const END_POINT = 'https://data-platform.mlit.go.jp/api/v1/';
// 取得したAPIキーを''内に入れてください
const API_KEY = 'please use your own key';


// 実際にクエリをポストする
// queryStr: クエリ内容
// queryName: クエリ名
export async function PostQuery(queryContents, queryName) {
  // HTTPクエリー内容を作成
  const request = {
    url: END_POINT,
    method: 'post',
    headers: {
      'Content-type': 'application/json',
      apikey: API_KEY
    },
    data: { query: queryContents }
  };

  // APIを呼び出して結果を準備する
  const result = await axios(request)
    .then((response) => response.data.data[queryName])
    .catch((error) => {
      console.error(error.message);
      if (error.response) console.debug('Error data:', error.response.data);
      return [];
    });

  return result;
}
python
# 通信に必要なライブラリのインポート
import requests

# エンドポイントとAPIキーを定義しておく
END_POINT = 'https://data-platform.mlit.go.jp/api/v1/'
API_KEY = 'please use your own key'

# 実際にクエリをポストする
# queryStr: クエリ内容
# queryName: クエリ名
async def post_query(queryContents, queryName):
    # HTTPクエリー内容を作成
    request = {
        "url": END_POINT,
        "method": "post",
        "headers": {
            "Content-type": "application/json",
            "apikey": API_KEY
        },
        "data": {"query": queryContents}
    }

    # APIを呼び出して結果を準備する
    try:
        response = requests.post(request["url"], headers=request["headers"], json=request["data"])
        response.raise_for_status()
        result = response.json()["data"][queryName]
    except requests.exceptions.RequestException as error:
        print(f"Error data: {error}")
        result = []

    return result
